# Runtime dependencies for `main.py --mode api`.
# Camera-only code paths (ATTENDANCE_CAMERA_INDEX, /attendance/live/*) are not
# needed on Render — the frontend sends captured frames as base64 over HTTP.

fastapi==0.115.6
uvicorn==0.32.1
python-dotenv==1.0.1

# MongoDB (async driver + fallback used in main.py)
motor==3.6.0
pymongo==4.9.2

# Vision / face recognition
numpy==1.26.4
opencv-python-headless==4.10.0.84
Pillow==10.4.0
# dlib has no official PyPI wheels and compiles from source (needs cmake +
# a C++ toolchain, several minutes, >1GB RAM) — too fragile for Render's
# build environment. dlib-bin ships the same binary prebuilt, no compile step.
dlib-bin==20.0.1
# face_recognition_models uses pkg_resources (from setuptools) at import time
# to locate its .dat model files. Modern venvs don't guarantee setuptools is
# preinstalled, so without this the import throws ModuleNotFoundError, which
# face_recognition's own except-block swallows and reports as a misleading
# "please install face_recognition_models" message even though it IS installed.
setuptools==75.6.0
face_recognition_models==0.3.0

# face_recognition==1.3.0 is intentionally NOT listed here: its metadata
# hard-requires "dlib" by name, which would make pip compile the real dlib
# from source even with dlib-bin already installed above. The Render build
# command installs it separately with --no-deps so it just uses the deps
# already satisfied in this file instead of chasing that requirement:
#   pip install -r requirements.txt && pip install --no-deps face_recognition==1.3.0
